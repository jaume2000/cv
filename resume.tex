% Header: photo left; name/contact centered. Plain-text contact line (no FontAwesome
% icons) so ATS extract clean phone/email/links instead of private-use glyphs.
\makeatletter
\renewcommand*{\makecvheader}[1][C]{%
  \newcommand*{\drawphoto}{%
    \ifthenelse{\isundefined{\@photo}}{}{%
      \newlength{\photodim}%
      \ifthenelse{\equal{\@photoshape}{circle}}%
        {\setlength{\photodim}{1.05cm}}%
        {\setlength{\photodim}{1.5cm}}%
      \ifthenelse{\equal{\@photoedge}{edge}}%
        {\def\@photoborder{darkgray}}%
        {\def\@photoborder{none}}%
      \begin{tikzpicture}%
        \node[\@photoshape, draw=\@photoborder, line width=0.3mm,
          inner sep=\photodim, fill overzoom image=\@photo] () {};
      \end{tikzpicture}%
    }%
  }%
  \newlength{\headerphotowidth}%
  \newlength{\headertextwidth}%
  \ifthenelse{\isundefined{\@photo}}{%
    \setlength{\headerphotowidth}{0cm}%
    \setlength{\headertextwidth}{\textwidth}%
  }{%
    \setlength{\headerphotowidth}{2.2cm}%
    \setlength{\headertextwidth}{\dimexpr\textwidth-2\headerphotowidth\relax}%
  }%
  \noindent
  \begin{minipage}[t]{\headerphotowidth}%
    \vspace{0pt}%
    \raggedright\drawphoto
  \end{minipage}%
  \begin{minipage}[t]{\headertextwidth}%
    \vspace{0pt}%
    \centering
    \headerfirstnamestyle{\@firstname}\headerlastnamestyle{{}\acvHeaderNameDelim\@lastname}%
    \\[\acvHeaderAfterNameSkip]%
    \ifthenelse{\isundefined{\@position}}{}{\headerpositionstyle{\@position\\[\acvHeaderAfterPositionSkip]}}%
    \ifthenelse{\isundefined{\@address}}{}{\headeraddressstyle{\@address\\[\acvHeaderAfterAddressSkip]}}%
    \headersocialstyle{%
      \newbool{isstart}%
      \setbool{isstart}{true}%
      \ifthenelse{\isundefined{\@mobile}}{}{%
        \@mobile%
        \setbool{isstart}{false}%
      }%
      \ifthenelse{\isundefined{\@email}}{}{%
        \ifbool{isstart}{\setbool{isstart}{false}}{\acvHeaderSocialSep}%
        \href{mailto:\@email}{\@email}%
      }%
      \ifthenelse{\isundefined{\@homepage}}{}{%
        \ifbool{isstart}{\setbool{isstart}{false}}{\acvHeaderSocialSep}%
        \href{https://\@homepage}{\@homepage}%
      }%
      \\[\acvHeaderAfterNameSkip]%
      \setbool{isstart}{true}%
      \ifthenelse{\isundefined{\@github}}{}{%
        \ifbool{isstart}{\setbool{isstart}{false}}{\acvHeaderSocialSep}%
        \href{https://github.com/\@github}{github.com/\@github}%
      }%
      \ifthenelse{\isundefined{\@linkedin}}{}{%
        \ifbool{isstart}{\setbool{isstart}{false}}{\acvHeaderSocialSep}%
        \href{https://www.linkedin.com/in/\@linkedin}{linkedin.com/in/\@linkedin}%
      }%
      \ifthenelse{\isundefined{\@googlescholarid}}{}{%
        \ifbool{isstart}{\setbool{isstart}{false}}{\acvHeaderSocialSep}%
        \href{https://scholar.google.com/citations?user=\@googlescholarid}{\@googlescholarname}%
      }%
    }\\[\acvHeaderAfterSocialSkip]%
  \end{minipage}%
  \begin{minipage}[t]{\headerphotowidth}%
    \vspace{0pt}%
    \hfill
  \end{minipage}%
  \par
}
